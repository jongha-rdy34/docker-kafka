cat <<EOF > system.log
[INFO] Start process
[ERROR] Disk read failure
[WARNING] Memory usage high
[INFO] Normal operation
[ERROR] Network timeout
[WARNING] CPU temperature warning
EOF

date=$(date, +%Y%m%d)
backup_path="backup_${date}"
mkdir -p "${backup_path}"
grep "ERROR" system.log > error.log
grep "WARNING" system.log > warning.log
echo "Log summary ${date}" > "${backup_path}/log_summary.txt"
cat error.log >> "${backup_path}/log_summary.txt"
cat warning.log >> "${backup_path}/log_summary.txt"
cat "${backup_path}/log_summary.txt"
