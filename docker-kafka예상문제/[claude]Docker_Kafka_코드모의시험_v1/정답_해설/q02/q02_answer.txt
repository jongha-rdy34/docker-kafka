# Q02 Docker 기본 명령 (8점)
# ______ 를 채워 완성된 명령/답을 작성하세요. 명령은 한 줄에 하나씩 씁니다.
# (컨테이너 내부) 표시가 있는 줄은 컨테이너 안의 셸에서 입력하는 명령입니다.

# [1] hello-world 이미지를 hello-test 라는 이름의 컨테이너로 실행
docker run --name hello-test hello-world

# [2] 종료(Exited)된 컨테이너까지 모두 조회
docker ps -a

# [3] hello-test 컨테이너 삭제
docker rm hello-test

# [4] 로컬에 내려받은 이미지 목록 조회
docker images

# [5] python:3.10-slim 이미지로 devbox 컨테이너를 터미널과 상호작용 가능한 모드로 실행하고 bash로 진입
docker run -it --name devbox python:3.10-slim bash

# [6] (컨테이너 내부) 현재 디렉토리 확인 → note.txt에 "inside devbox" 저장 → 내용 확인 → 6*7 출력 → 종료
pwd
echo "inside devbox" > note.txt
cat note.txt
python -c "print(6 * 7)"
exit

# [7] 종료된 devbox를 다시 시작하면서 바로 터미널에 연결
docker start -ai devbox

# [8] (컨테이너 내부) note.txt 확인 후 종료, 이어서 호스트에서 devbox 중지·삭제
cat note.txt
exit
docker stop devbox
docker rm devbox

# [9] [8]에서 note.txt가 남아 있는 이유와, [8]에서 삭제한 뒤 [5]를 다시 실행하면 note.txt가 없는 이유를 쓰세요.
답: docker start는 기존 컨테이너를 다시 실행하므로 그 컨테이너의 쓰기 계층(파일 변경)이 유지된다. docker rm으로 컨테이너를 지우면 쓰기 계층도 사라지고, docker run은 이미지로부터 새 컨테이너를 만들기 때문에 note.txt가 없다.
