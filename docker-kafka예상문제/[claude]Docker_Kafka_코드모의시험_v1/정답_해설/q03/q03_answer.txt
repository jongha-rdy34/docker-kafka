# Q03 Flask 앱 이미지 빌드·실행 (12점) — 명령 부분
# 모든 명령은 q03 폴더 안에서 실행합니다.

# [1] 현재 디렉토리의 Dockerfile로 delivery-api 이미지 빌드
docker build -t delivery-api .

# [2] delivery-api 이미지를 백그라운드로 실행
#     호스트 8000 포트 → 컨테이너 5000 포트, 컨테이너 이름 delivery-api-container
docker run -d -p 8000:5000 --name delivery-api-container delivery-api

# [3] 실행 중인 컨테이너 확인
docker ps

# [4] 호스트에서 두 경로 요청
curl http://localhost:8000/
curl http://localhost:8000/status

# [5] 컨테이너 중지 후 삭제
docker stop delivery-api-container
docker rm delivery-api-container
